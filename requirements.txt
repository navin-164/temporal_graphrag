# Install:
#   conda activate temporal_rag
#   pip install -r requirements.txt

# PyTorch note: `torch` below resolves to whatever PyPI serves your platform,
# which is CPU-only by default on most Windows machines — that's the right
# choice unless you have an NVIDIA GPU with CUDA already set up. If you do,
# install torch FIRST with the command from pytorch.org/get-started/locally
# (pick your CUDA version), THEN run this file — pip won't touch a
# requirement that's already satisfied.

# ---- Embeddings & local ML models (run on your machine — no API calls, no rate limits) ----
# SPECTER2, SciBERT, CLIP, TAPAS, and BLIP-2 all load through transformers /
# sentence-transformers directly. CORRECTION from the original version of
# this file: SPECTER2's own model card explicitly says its base encoder
# alone is NOT what to use for general embeddings — "please use
# allenai/specter2" instead, which is actually an adapter (the retrieval/
# proximity one) loaded onto the base model via the `adapters` package.
# Skipping `adapters`, as originally planned here, would have run without
# any error and just silently produced lower-quality embeddings than
# SPECTER2's own documentation recommends — so it's back in, not optional.
torch>=2.3                    # backend for transformers/sentence-transformers — see PyTorch note above
transformers>=4.42
sentence-transformers>=3.0
adapters>=1.0.0                # required for SPECTER2's recommended usage — see embeddings/text_embedder.py
Pillow>=10.0                    # image loading for embeddings/figure_embedder.py — listed explicitly
                                 # rather than relying on it arriving transitively via matplotlib/streamlit
accelerate>=0.31              # efficient model loading (device placement, lower memory use)
sentencepiece>=0.2.0          # tokenizer backend several HF models need but don't bundle

# ---- Document parsing ----
lxml>=5.2                     # parsing GROBID's TEI XML output
PyMuPDF>=1.24                 # (imported as `fitz`) fallback text extraction if GROBID errors on a PDF

# ---- Graph algorithms (Python-side, complements Neo4j) ----
networkx>=3.3

# ---- Core data stack ----
numpy>=1.26
pandas>=2.2

# ---- Reliability & performance ----
tenacity>=8.4                 # retry/backoff decorators for flaky external calls (Groq/Gemini/GROBID/Neo4j)
diskcache>=5.6                # local disk cache — avoids recomputing embeddings and re-spending
                               # free-tier LLM quota every time you re-run a script during development
structlog>=24.2                # structured (JSON-capable) logging, per the project's logging requirement
